--- Example 1: Letter-style (TAN Dismissal style) ---
{\rtf1\ansi\deff0
{\fonttbl{\f0 Times New Roman;}}
\f0\fs24
\par
@[Subdocument(3)]
\par
\par
@[builtin.today.FormatDate(preset1)]
\par
@[If(@[Defense.Any()] == true)]
@[Respondent.first.FormatName(F L).SetCasing(Title)]
\par
c/o @[Defense.first.FormatName(F L).SetCasing(Title)], Esq.
\par
@[DefenseAddress.first.StreetAddress.SetCasing(Title)]
\par
@[DefenseAddress.first.City.SetCasing(Title)], @[DefenseAddress.first.State] @[DefenseAddress.first.Zip]
\par
@[Else]
@[RespondentAddress.ForEach(x)]
@[Respondent.first.FormatName(F L).SetCasing(Title)]
\par
@[RespondentAddress.first.StreetAddress.SetCasing(Title)]
\par
@[RespondentAddress.first.City.SetCasing(Title)], @[RespondentAddress.first.State] @[RespondentAddress.first.Zip]
\par
@[RespondentAddress.EndForEach]
@[EndIf]
\par
Case Numbers: @[ProsNum.first.Number]
\par
\par
Dear @[If(@[Defense.Any()] == true)]@[DefenseInfo.Prefix] @[Defense.first.FormatName(L).SetCasing(Title)]@[Else]@[If(@[RespondentInfo.Gender] == M)]Mr.@[ElseIf(@[RespondentInfo.Gender] == F)]Ms.@[Else]Mr./Ms.@[EndIf] @[Respondent.first.FormatName(L).SetCasing(Title)]@[EndIf]:
\par
\par
Body paragraph with fillpoints embedded in formatted text. Use @[ ... ] for names, dates, addresses, and conditionals.
\par
\par
Sincerely,
\par
@[cu.FormatName(F L)]
}
